if not SUPABASE_URL or not SUPABASE_KEY:
    raise ValueError("Missing Supabase environment variables!")

supabase: Client = create_client(SUPABASE_URL, SUPABASE_KEY)

class handler(BaseHTTPRequestHandler):
    def do_GET(self):
        today = datetime.date.today().strftime("%d/%m/%Y")

        response = supabase.table("tasks").select("*").eq("date", today).execute()
        tasks = response.data if response.data else []

        self.send_response(200)
        self.send_header("Content-Type", "application/json")
        self.end_headers()
        self.wfile.write(json.dumps(tasks).encode())

# api/save_tasks.py
import os
import json
import datetime
import sys
from http.server import BaseHTTPRequestHandler
from supabase import create_client, Client
# Consider importing PostgrestAPIResponse if using supabase-py v1 for detailed error checks
# from supabase import PostgrestAPIResponse

# --- Logging Setup ---
def log(message):
    print(message, file=sys.stderr)
    sys.stderr.flush()
# --- End Logging Setup ---

SUPABASE_URL = os.getenv("SUPABASE_URL")
# This script modifies data, safer to use the SERVICE_ROLE key
SUPABASE_KEY = os.getenv("SUPABASE_KEY")

log(f"save_tasks: SUPABASE_URL defined: {bool(SUPABASE_URL)}")
log(f"save_tasks: SUPABASE_KEY defined: {bool(SUPABASE_KEY)}")

if not SUPABASE_URL or not SUPABASE_KEY:
    log("ERROR: save_tasks: Missing Supabase environment variables!")
    raise ValueError("Missing Supabase environment variables!")

try:
    # Ensure SERVICE_ROLE key is used if you want to bypass RLS for this script
    supabase = create_client(SUPABASE_URL, SUPABASE_KEY)
    log("save_tasks: Supabase client created successfully")
except Exception as e:
    log(f"ERROR creating Supabase client in save_tasks: {str(e)}")
    raise

class handler(BaseHTTPRequestHandler):
    def do_POST(self):
        log("==== save_tasks POST request received ====")
        content_length = int(self.headers.get('Content-Length', 0))
        log(f"Content length: {content_length}")

        post_data = self.rfile.read(content_length)
        log(f"Raw post data: {post_data}")

        try:
            tasks = json.loads(post_data)
            log(f"Parsed tasks: {json.dumps(tasks)}")
        except Exception as e:
            log(f"ERROR parsing JSON in save_tasks: {str(e)}")
            self.send_response(400)
            self.send_header("Content-Type", "application/json")
            self.send_header("Access-Control-Allow-Origin", "*") # CORS
            self.end_headers()
            self.wfile.write(json.dumps({"error": f"Invalid JSON: {str(e)}"}).encode())
            return

        today = datetime.date.today().strftime("%d/%m/%Y") # Or "%Y-%m-%d" if using DATE type
        log(f"Today's date: {today}")

        try:
            # Process all tasks at once for better efficiency
            rows = []
            for i, task in enumerate(tasks):
                task_name = task.get("task_name")
                elapsed_time = task.get("elapsed_time")

                log(f"Processing task {i+1}: name='{task_name}', time='{elapsed_time}'")

                if not task_name:
                    log(f"Skipping task {i+1} due to empty name")
                    continue

                rows.append({
                    "date": today,
                    "task_name": task_name,
                    "elapsed_time": elapsed_time,
                    # Use timezone-aware UTC time for created_at
                    "created_at": datetime.datetime.now(datetime.timezone.utc).isoformat()
                })

            log(f"Prepared {len(rows)} rows for task insertion")

            task_insert_success = False
            if rows:
                log("Attempting to delete existing tasks for today first...")
                # Delete existing tasks for today before inserting new ones
                delete_result = supabase.table("tasks").delete().eq("date", today).execute()
                log(f"Task delete result: {delete_result}")
                # Add specific error checks for delete_result if needed

                log("Now inserting new tasks...")
                # Simple insert without upsert
                insert_result = supabase.table("tasks").insert(rows).execute()
                log(f"Task insert result: {insert_result}")

                # --- Proper error checking for insert_result needed here ---
                # Example check (adapt based on supabase-py version):
                # if isinstance(insert_result, PostgrestAPIResponse) and insert_result.error:
                #    log(f"ERROR in task insert: {insert_result.error}")
                #    raise Exception(f"Supabase task insert error: {insert_result.error.message}")
                # elif check other potential failure modes...
                # else:
                #    task_insert_success = True
                task_insert_success = True # Assume success for now if no exception/error object found

                log("Task insert operation completed.")
            else:
                log("No valid task rows to insert, skipping task delete/insert.")
                # Treat as success for metadata update if no tasks needed saving
                task_insert_success = True


            # **** START METADATA UPDATE ****
            metadata_update_error = None
            if task_insert_success: # Only update metadata if task save was okay
                log("Task save considered successful. Now updating last backup time.")
                try:
                    # Get current time in ISO format with timezone
                    last_backup_iso_time = datetime.datetime.now(datetime.timezone.utc).isoformat()

                    # Use upsert for robustness: insert if not exists, update if exists
                    meta_update_result = supabase.table("metadata").upsert({
                        "key": "last_backup_time", # Specify the key
                        "value": last_backup_iso_time,
                        # updated_at is handled by DB default trigger now
                    }).execute()

                    log(f"Metadata update/upsert result: {meta_update_result}")
                    # Add specific error checks for meta_update_result if needed

                    log("Last backup time updated successfully.")

                except Exception as meta_e:
                    metadata_update_error = str(meta_e)
                    log(f"ERROR updating last backup time in metadata: {metadata_update_error}")
                    # Decide if this error should prevent a 200 OK response - maybe not?
            # **** END METADATA UPDATE ****


            log("Sending success response")
            self.send_response(200)
            self.send_header("Content-Type", "application/json")
            self.send_header("Access-Control-Allow-Origin", "*") # CORS
            self.end_headers()
            response_body = {
                "message": "Tasks saved successfully",
                "count": len(rows)
            }
            if metadata_update_error:
                 response_body["metadata_warning"] = f"Failed to update metadata: {metadata_update_error}"
            self.wfile.write(json.dumps(response_body).encode())

        except Exception as e:
            log(f"ERROR in save_tasks processing loop: {str(e)}")
            self.send_response(500)
            self.send_header("Content-Type", "application/json")
            self.send_header("Access-Control-Allow-Origin", "*") # CORS
            self.end_headers()
            self.wfile.write(json.dumps({
                "error": f"Failed to save tasks: {str(e)}"
            }).encode())

    # Handle CORS preflight requests
    def do_OPTIONS(self):
        self.send_response(204)
        self.send_header("Access-Control-Allow-Origin", "*")
        self.send_header("Access-Control-Allow-Methods", "POST, OPTIONS") # Allow POST
        self.send_header("Access-Control-Allow-Headers", "Content-Type")
        self.end_headers()
